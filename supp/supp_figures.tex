\begin{figure}[h!]
    \centering
    \includegraphics[width=\textwidth]{FigS_gap}
    \caption{The measured variance between the two equilibrium ends, one row per population on a common $V_A$ axis. Blue diamonds: the variance mutation--selection--drift balance maintains, $U_d\bar{s}$ (strong-selection limit; Appendix~D), with $U_d$ from the direct mutation rate and $\bar{s}$ the mean of the empirical DFE assigned to the population; where the assignment is an equal mixture of two species, the band spans $U_d$ times their means. Circles and bars: measured $\hat{V}_A$, posterior mode, 50\% (thick) and 95\% (thin) HPD. Orange diamonds: the least variance that could account for the observed deficit at unlinked sites, $\ln(1/\delta)/(4\kappa(0))$ (Equation~\eqref{eq:vamax_mult}), from $\pi$ of the study species (filled) or of another population of the species (open), with bars over the 95\% CI of the mutation rate. The superb fairy-wren has no annotated assembly; Soay sheep have no usable census.}
    \label{fig:S_gap}
\end{figure}

\begin{figure}[h!]
    \centering
    \includegraphics[width=0.85\textwidth]{FigS_rescaling_linked}
    \caption{As Figure~\ref{fig:rescaling}, with linkage, for the six populations with both an observed deficit and a linkage map (collared flycatchers, rhesus macaques, song sparrows, bighorn sheep and both great tit populations). Blue: $\bar{V}_A/(U_d\bar{s})$. Orange dashed: predicted relative to observed diversity, unlinked, under the multiplicative upper bound. Teal: the same with linkage, multiplied by $\overline{\Omega}(\bar{V}_A)$ at the same assigned DFE, so that both comparisons share one distribution of effects. Open circle: the average at $c = U_d\bar{s}/\hat{V}_A$, where the variance is what that DFE maintains, which is the self-consistent prediction of mutation--selection balance with background selection. Bold: species averages of the $\log_{10}$ ratio (great tits averaged first); faint: populations.}
    \label{fig:S_rescaling_linked}
\end{figure}
